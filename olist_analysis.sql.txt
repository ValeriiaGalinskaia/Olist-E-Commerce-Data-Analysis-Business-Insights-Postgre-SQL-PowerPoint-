-- ----------------------------------------------------------------------------
--1. EDA & DATA DISCOVERY
-- ----------------------------------------------------------------------------
SELECT * 
FROM orders
LIMIT 20;
SELECT 
    MIN(o.order_purchase_timestamp)::date AS first_order_date,
    MAX(o.order_purchase_timestamp)::date AS last_order_date,
    COUNT(o.order_id) AS total_orders,
    COUNT(DISTINCT c.customer_unique_id) AS unique_customers,
    ROUND(
        100.0 * COUNT(CASE WHEN order_status = 'delivered' THEN 1 END) / COUNT(order_id), 
        2
    ) AS delivered_percentage
FROM orders o
JOIN customers c ON o.customer_id = c.customer_id;

-- ----------------------------------------------------------------------------
-- 2. FINANCE AND DYNAMICS: Monthly revenue and average order value (AOV)
-- ----------------------------------------------------------------------------

SELECT 
    DATE_TRUNC('month', o.order_purchase_timestamp)::date AS order_month,
    COUNT(DISTINCT o.order_id) AS total_orders,
    ROUND(SUM(p.payment_value)::numeric, 2) AS total_revenue,
    ROUND(AVG(p.payment_value)::numeric, 2) AS average_order_value
FROM orders o
JOIN order_payments p ON o.order_id = p.order_id
WHERE o.order_status = 'delivered' -- считаем деньги только по доставленным заказам
GROUP BY order_month
ORDER BY order_month;

-- ----------------------------------------------------------------------------
-- 3. LOGISTICS AND CUSTOMER EXPERIENCE: The impact of delivery delays on ratings (Reviews)
-- ----------------------------------------------------------------------------

Логистика
SELECT 
    ROUND(AVG(EXTRACT(EPOCH FROM (order_delivered_customer_date - order_purchase_timestamp)) / 86400)::numeric, 1) AS avg_actual_delivery_days,
    ROUND(AVG(EXTRACT(EPOCH FROM (order_estimated_delivery_date - order_purchase_timestamp)) / 86400)::numeric, 1) AS avg_estimated_delivery_days
FROM orders
WHERE order_status = 'delivered'
  AND order_delivered_customer_date IS NOT NULL;

SELECT 
    CASE 
        WHEN order_delivered_customer_date > order_estimated_delivery_date THEN 'Late'
        ELSE 'On Time / Early'
    END AS delivery_status,
    COUNT(o.order_id) AS total_orders,
    ROUND(100.0 * COUNT(o.order_id) / SUM(COUNT(o.order_id)) OVER(), 2) AS percentage,
    ROUND(AVG(r.review_score)::numeric, 2) AS avg_review_score
FROM orders o
JOIN order_reviews r ON o.order_id = r.order_id
WHERE o.order_status = 'delivered'
  AND o.order_delivered_customer_date IS NOT NULL
GROUP BY delivery_status;


-- ----------------------------------------------------------------------------
-- 4. PRODUCT MATRIX: Top 10 categories by revenue volume
-- ----------------------------------------------------------------------------

SELECT 
    p.product_category_name AS category,
    COUNT(DISTINCT i.order_id) AS total_orders,
    COUNT(i.product_id) AS items_sold,
    ROUND(SUM(i.price)::numeric, 2) AS total_revenue,
    ROUND(AVG(i.price)::numeric, 2) AS avg_item_price
FROM order_items i
JOIN orders o ON i.order_id = o.order_id
JOIN products p ON i.product_id = p.product_id
WHERE o.order_status = 'delivered'
  AND p.product_category_name IS NOT NULL
GROUP BY p.product_category_name
ORDER BY total_revenue DESC
LIMIT 10;

-- ----------------------------------------------------------------------------
-- 4. GEOGRAPHY: Distribution of orders by customer states (Top-5)
-- ----------------------------------------------------------------------------


SELECT 
    c.customer_state AS state,
    COUNT(DISTINCT o.order_id) AS total_orders,
    ROUND(SUM(p.payment_value)::numeric, 2) AS total_revenue,
    ROUND(
        100.0 * COUNT(DISTINCT o.order_id) / SUM(COUNT(DISTINCT o.order_id)) OVER(), 
        2
    ) AS orders_share_percentage
FROM orders o
JOIN customers c ON o.customer_id = c.customer_id
JOIN order_payments p ON o.order_id = p.order_id
WHERE o.order_status = 'delivered'
GROUP BY c.customer_state
ORDER BY total_orders DESC
LIMIT 5;
